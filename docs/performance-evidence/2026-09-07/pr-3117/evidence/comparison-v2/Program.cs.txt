using BenchmarkDotNet.Configs;
using BenchmarkDotNet.Jobs;
using BenchmarkDotNet.Running;
using Perfolizer.Horology;
var root = "C:/git/Dekaf-perf-improvements-20260907/pr-3117";
var dry = args.Contains("--dry");
var profile = args.Contains("--profile");
var arguments = args.Where(a => a is not ("--dry" or "--profile")).ToArray();
var artifactIndex = Array.IndexOf(arguments, "--artifacts");
var basis = (dry ? Job.Dry : Job.Default.WithWarmupCount(profile ? 3 : 5).WithIterationCount(profile ? 3 : 15).WithIterationTime(TimeInterval.FromMilliseconds(250)))
    .WithAffinity(new IntPtr(4)).WithEnvironmentVariable("DOTNET_TieredCompilation", "0")
    .WithEnvironmentVariable("DEKAF_DISPATCH_PROFILE", profile ? "1" : "0")
    .WithEnvironmentVariable("DEKAF_DISPATCH_SAMPLE_DIRECTORY", Path.Combine(arguments[artifactIndex + 1], "latency-samples"));
var config = DefaultConfig.Instance.WithOptions(ConfigOptions.KeepBenchmarkFiles);
foreach (var (name, directory) in new[] { ("A_Predecessor", "predecessor"), ("B_Published", "published"), ("C_SingleProbe", "candidate"), ("D_ClearOnce", "candidate-v2"), ("E_PublishedControl", "published"), ("F_PredecessorControl", "predecessor") })
    config = config.AddJob(basis.WithId(name).WithEnvironmentVariable("DEKAF_DISPATCH_JOB", name)
        .WithMsBuildArguments($"/p:ProductDirectory={root}/{directory}", "/p:UseSharedCompilation=false"));
var summaries = BenchmarkSwitcher.FromAssembly(typeof(Program).Assembly).Run(arguments, config).ToArray();
return summaries.Length == 0 || summaries.Any(s => s.HasCriticalValidationErrors || s.Reports.Length == 0
    || s.Reports.Any(r => !r.Success || r.ResultStatistics is null)) ? 1 : 0;
