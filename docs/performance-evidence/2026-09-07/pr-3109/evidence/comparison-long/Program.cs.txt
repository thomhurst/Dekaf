using BenchmarkDotNet.Configs;
using BenchmarkDotNet.Exporters.Json;
using BenchmarkDotNet.Jobs;
using BenchmarkDotNet.Running;
using Perfolizer.Mathematics.OutlierDetection;

var dry = args.Contains("--dry");
var job = (dry ? Job.Dry : Job.Default.WithWarmupCount(30).WithIterationCount(300).WithLaunchCount(2))
    .WithOutlierMode(OutlierMode.DontRemove)
    .WithAffinity(new IntPtr(4))
    .WithEnvironmentVariable("DOTNET_TieredCompilation", "0");
var root = "C:/git/Dekaf-perf-improvements-20260907/pr-3109";
var config = DefaultConfig.Instance.WithOptions(ConfigOptions.KeepBenchmarkFiles)
    .AddExporter(JsonExporter.Full);
foreach (var (name, directory) in new[] { ("A_Published", "published"), ("B_Candidate", "candidate"), ("C_PublishedControl", "published") })
    config = config.AddJob(job.WithId(name).WithMsBuildArguments($"/p:ProductDirectory={root}/{directory}", "/p:UseSharedCompilation=false"));
var summaries = BenchmarkSwitcher.FromAssembly(typeof(Program).Assembly)
    .Run(args.Where(a => a != "--dry").ToArray(), config).ToArray();
return summaries.Length == 0 || summaries.Any(s => s.HasCriticalValidationErrors || s.Reports.Length == 0 || s.Reports.Any(r => !r.Success || r.ResultStatistics is null)) ? 1 : 0;
